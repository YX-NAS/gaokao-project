### 3.2 推荐算法模块

```javascript
// 核心推荐函数
function calculateRecommendations(score, subjects, data, year) {
  // 1. 获取考生位次
  const studentRank = getRank(score, year);

  // 2. 遍历所有院校
  data.forEach(school => {
    // 2.1 过滤选科不符的
    if (!matchSubjects(school, subjects)) return;

    // 2.2 获取院校录取位次
    const schoolRank = getRank(school.minScore, year);

    // 2.3 计算位次差
    const rankDiff = studentRank - schoolRank;

    // 2.4 分类
    if (rankDiff > 0 && rankDiff <= schoolRank * 0.08) {
      // 冲
    } else if (rankDiff <= 0 && rankDiff >= -schoolRank * 0.07) {
      // 稳
    } else if (rankDiff < -schoolRank * 0.07) {
      // 保
    }
  });

  return { chong, wen, bao };
}
```

### 3.3 位次查询模块

```javascript
// 二分查找优化
function getRank(score, year) {
  const segments = scoreSegments[year];
  const scores = Object.keys(segments)
    .map(Number)
    .sort((a, b) => b - a);  // 降序排列

  // 二分查找
  let left = 0, right = scores.length - 1;
  while (left <= right) {
    const mid = Math.floor((left + right) / 2);
    if (scores[mid] <= score) {
      return segments[scores[mid]];
    } else {
      right = mid - 1;
    }
  }
  return segments[scores[scores.length - 1]];
}
```

// LocalStorage 结构
const VOLUNTEER_KEY = 'gaokao_volunteer_plans';
const ACTIVE_PLAN_KEY = 'gaokao_active_plan';

function savePlan(plan) {
  const plans = JSON.parse(localStorage.getItem(VOLUNTEER_KEY) || '[]');
  const existing = plans.findIndex(p => p.id === plan.id);
  if (existing >= 0) {
    plans[existing] = plan;
  } else {
    plans.push(plan);
  }
  localStorage.setItem(VOLUNTEER_KEY, JSON.stringify(plans));
}

## 六、性能优化

### 6.1 静态资源

- 所有资源 gzip 压缩
- 静态文件设置长期缓存（1年）
- 数据文件按年份拆分，按需加载

### 6.2 运行时

- 推荐算法时间复杂度 O(n log n)（排序）
- 数据量 ~4000条，浏览器无压力
- 使用 DocumentFragment 减少 DOM 操作

### 6.3 CDN加速

- 静态资源走 CDN
- 数据文件走 CDN
- 域名解析使用 CDN 智能调度

## 七、安全设计

### 7.1 前端安全

- 无后端，无数据库，无注入风险

- 全站 HTTPS
